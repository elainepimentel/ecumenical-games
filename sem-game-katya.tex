\item[$(\mathbf{P}_\wedge)$] At $\mathbf{P}, w: \phi_1\wedge \phi_2$, \You
    choose between $\mathbf{P},w:\phi_1$ and 
    $\mathbf{P},w:\phi_2$ to continue the game. 
    
\item[$(\mathbf{O}_\wedge)$] At $\mathbf{O}, w:
    \phi_1\wedge \phi_2$, \Ic choose between $\mathbf{O}, w:\phi_1$
    and $\mathbf{O}, w:\phi_2$ to continue the game.

\item[$(\mathbf{P}_{\vee_i})$] At $\mathbf{P}, w: \phi_1 \vee_i \phi_2$, \Ic choose between $\mathbf{P}, w:\phi_1$ and $\mathbf{P}, w:\phi_2$ to continue the game.

\item[$(\mathbf{O}_{\vee_i})$] At $\mathbf{O}, w: \phi_1 \vee_i \phi_2$, \You choose between $\mathbf{O}, w:\phi_1$ and $\mathbf{O}, w:\phi_2$ to continue the game.

\vspace{2mm}

\item[$(\mathbf{P}_{\vee_c})$] At $\mathbf{P}, w: \phi_1 \vee_c \phi_2$, \You choose $v \geq w$, \Ic choose $u \geq v$ and between $\mathbf{P}, u:\phi_1$ and $\mathbf{P}, u:\phi_2$ to continue the game.

\item[$(\mathbf{O}_{\vee_c})$] At $\mathbf{O}, w: \phi_1 \vee_c \phi_2$, \Ic choose $v \geq w$, \You choose $u \geq v$ and between $\mathbf{O}, u:\phi_1$ and $\mathbf{O}, u:\phi_2$ to continue the game.

\vspace{2mm}

\item[$(\mathbf{P}_\neg)$] At $\mathbf{P}, w: \neg \phi$, \You choose $v \geq w$ and the game continues with $\mathbf{O}, v: \phi$.

\item[$(\mathbf{O}_\neg)$] At $\mathbf{O}, w: \neg \phi$, \Ic choose $v \geq w$ and the game continues with $\mathbf{P}, v: \phi$.

\vspace{2mm}

\item[$(\mathbf{P}_{\iimp})$] At $\mathbf{P}, w: \phi_1 \iimp \phi_2$, \You choose $v \geq w$, \Ic choose between $\mathbf{O}, v:\phi_1$ and $\mathbf{P}, v:\phi_2$ to continue the game.

\item[$(\mathbf{O}_{\iimp})$] At $\mathbf{O}, w: \phi_1 \iimp \phi_2$, \Ic choose $v \geq w$, \You choose between $\mathbf{P}, v:\phi_1$ and $\mathbf{O}, v:\phi_2$ to continue the game.

\vspace{2mm}

\item[$(\mathbf{P}_{\cimp})$] At $\mathbf{P}, w: \phi_1 \cimp \phi_2$, \You choose $v \geq w$, \Ic choose between $\mathbf{O}, v:\phi_1$ and ($u \geq v$ and $\mathbf{P}, v:\phi_2$) to continue the game.

\item[$(\mathbf{O}_{\cimp})$] At $\mathbf{O}, w: \phi_1 \cimp \phi_2$, \Ic choose $v \geq w$, \You choose between $\mathbf{P}, v:\phi_1$ and ($u \geq v$ and $\mathbf{O}, u:\phi_2$) to continue the game.
